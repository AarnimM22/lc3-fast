cmake_minimum_required(VERSION 3.20.0)
set(CMAKE_OBJECT_PATH_MAX 240)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(benchmark)
set(BENCH_ROLE tx CACHE STRING "tx, tx54 or rx")
set(BENCH_CODEC etsi CACHE STRING "etsi or google")
set(BENCH_FRAMES 200 CACHE STRING "Frames per case")
set(BENCH_SOAK_FRAMES 0 CACHE STRING "Frames per transport soak case")
set(BENCH_BITRATE 320000 CACHE STRING "Total stereo bitrate")
set(BENCH_PCM_BITS 16 CACHE STRING "LC3plus input precision; Opus sweeps 16 and 24")
set(BENCH_ASYNC_ONLY 0 CACHE STRING "Run nRF54 buffered radio cases only")
set(BENCH_USB 0 CACHE STRING "Receive PCM through USB Audio Class")
set(BENCH_OPTIMIZATIONS 0 CACHE STRING "Optimization experiment bit mask")
zephyr_get(BENCH_ROLE SYSBUILD)
zephyr_get(BENCH_CODEC SYSBUILD)
zephyr_get(BENCH_FRAMES SYSBUILD)
zephyr_get(BENCH_SOAK_FRAMES SYSBUILD)
zephyr_get(BENCH_BITRATE SYSBUILD)
zephyr_get(BENCH_PCM_BITS SYSBUILD)
zephyr_get(BENCH_ASYNC_ONLY SYSBUILD)
zephyr_get(BENCH_USB SYSBUILD)
zephyr_get(BENCH_OPTIMIZATIONS SYSBUILD)
target_compile_definitions(app PRIVATE BENCH_OPTIMIZATIONS=${BENCH_OPTIMIZATIONS})
math(EXPR BENCH_LEAN "${BENCH_OPTIMIZATIONS} & 1")
message(STATUS "Bench codec=${BENCH_CODEC} frames=${BENCH_FRAMES} soak=${BENCH_SOAK_FRAMES} bitrate=${BENCH_BITRATE} pcm_bits=${BENCH_PCM_BITS} LTO=${CONFIG_LTO}")
if(BENCH_CODEC MATCHES "-lto$" AND NOT CONFIG_LTO)
  message(FATAL_ERROR "Requested LTO was not enabled by Kconfig")
endif()
target_sources(app PRIVATE src/radio.c)
if(BENCH_USB)
  if(NOT BENCH_ROLE STREQUAL "tx" OR NOT BENCH_CODEC STREQUAL "google-custom-lto")
    message(FATAL_ERROR "USB benchmark requires tx/google-custom-lto")
  endif()
  target_sources(app PRIVATE src/main_usb.c src/usb_pcm.c src/usb_crc.c)
  target_compile_definitions(app PRIVATE BENCH_USB=1)
else()
  target_sources(app PRIVATE src/main.c)
endif()
target_compile_definitions(app PRIVATE BENCH_FRAMES=${BENCH_FRAMES} BENCH_SOAK_FRAMES=${BENCH_SOAK_FRAMES}
  BENCH_BITRATE=${BENCH_BITRATE} BENCH_PCM_BITS=${BENCH_PCM_BITS} BENCH_ASYNC_ONLY=${BENCH_ASYNC_ONLY})
if(BENCH_ROLE MATCHES "^tx")
  target_compile_definitions(app PRIVATE BENCH_TX=1)
  if(BENCH_ROLE STREQUAL "tx54")
    if(NOT BENCH_CODEC STREQUAL "google-lto")
      message(FATAL_ERROR "The tx54 benchmark currently supports google-lto")
    endif()
    target_compile_definitions(app PRIVATE BENCH_NRF54=1 BENCH_HAS_ASYNC=1)
    target_sources(app PRIVATE src/radio_async.c)
  endif()
  if(BENCH_CODEC MATCHES "^opus")
    include(opus.cmake)
  elseif(BENCH_CODEC MATCHES "^sbc")
    include(sbc.cmake)
  elseif(BENCH_CODEC MATCHES "^wavpack")
    include(wavpack.cmake)
  else()
    target_sources(app PRIVATE src/codec.c)
  endif()
  if(BENCH_CODEC MATCHES "^(opus|sbc|wavpack)")
    # Source list, precision and Cortex-M4 DSP options are in opus.cmake.
  elseif(BENCH_CODEC MATCHES "^etsi")
    set(REF ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/lc3plus/fixed_point)
    file(GLOB CODEC_SOURCES ${REF}/*.c ${REF}/basic_op/*.c)
    list(FILTER CODEC_SOURCES EXCLUDE REGEX "/(codec_exe|ccConvert)\\.c$")
    # The desktop reference puts a 200 kB dummy scratch array on the stack.
    # Reuse our statically allocated scratch area during initialization.
    # This changes storage only; the codec arithmetic is unchanged.
    list(REMOVE_ITEM CODEC_SOURCES ${REF}/lc3plus.c)
    file(READ ${REF}/lc3plus.c PORTED_SOURCE)
    string(REPLACE "UWord8 tmp_scratch[LC3PLUS_ENC_MAX_USER_SYSTEM_SCRATCH_SIZE];"
      "extern unsigned char bench_codec_scratch[]; UWord8 *tmp_scratch = bench_codec_scratch;"
      PORTED_SOURCE "${PORTED_SOURCE}")
    file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/lc3plus_port.c "${PORTED_SOURCE}")
    list(APPEND CODEC_SOURCES ${CMAKE_CURRENT_BINARY_DIR}/lc3plus_port.c)
    target_sources(app PRIVATE ${CODEC_SOURCES})
    target_sources(app PRIVATE src/ref_port.c)
    target_include_directories(app PRIVATE ${REF} ${REF}/basic_op)
    target_compile_definitions(app PRIVATE CODEC_ETSI=1 SUBSET_ALL DISABLE_HR_MODE WMOPS=0 DONT_COUNT_MEM NO_SCRATCH_STATS NDEBUG)
  else()
    set(REF ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/liblc3)
    file(GLOB CODEC_SOURCES ${REF}/src/*.c)
    if(BENCH_CODEC STREQUAL "google-custom-lto")
      set(CUSTOM_DIR ${CMAKE_CURRENT_BINARY_DIR}/lc3-custom)
      set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
        ${CMAKE_CURRENT_SOURCE_DIR}/../../tools/prepare-lc3-custom.py
        ${CMAKE_CURRENT_SOURCE_DIR}/../../tools/prepare-lc3-lite.py
        ${CMAKE_CURRENT_SOURCE_DIR}/src/custom_sns.inc
        ${CMAKE_CURRENT_SOURCE_DIR}/src/custom_spec.inc)
      execute_process(COMMAND ${PYTHON_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/../../tools/prepare-lc3-custom.py
        --source ${REF}/src --output ${CUSTOM_DIR}
        COMMAND_ERROR_IS_FATAL ANY)
      file(GLOB CODEC_SOURCES ${CUSTOM_DIR}/*.c)
      target_sources(app PRIVATE src/lc3_lite.c src/lc3_custom.c src/radio_async.c)
      target_include_directories(app PRIVATE src ${CUSTOM_DIR})
      target_compile_definitions(app PRIVATE CODEC_LC3_CUSTOM=1 CODEC_LC3_LITE=1 BENCH_HAS_ASYNC=1)
      if(NOT BENCH_LEAN)
        target_compile_definitions(app PRIVATE LC3_LITE_PROFILE=1)
      endif()
    elseif(BENCH_CODEC STREQUAL "google-lite-lto")
      set(LITE_DIR ${CMAKE_CURRENT_BINARY_DIR}/lc3-lite)
      set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
        ${CMAKE_CURRENT_SOURCE_DIR}/../../tools/prepare-lc3-lite.py
        ${REF}/src/lc3.c ${REF}/src/tns.c ${REF}/src/spec.c)
      execute_process(COMMAND ${PYTHON_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/../../tools/prepare-lc3-lite.py
        --source ${REF}/src --output ${LITE_DIR}
        COMMAND_ERROR_IS_FATAL ANY)
      foreach(unit lc3 tns spec)
        list(REMOVE_ITEM CODEC_SOURCES ${REF}/src/${unit}.c)
        list(APPEND CODEC_SOURCES ${LITE_DIR}/${unit}.c)
      endforeach()
      target_sources(app PRIVATE src/lc3_lite.c src/radio_async.c)
      target_include_directories(app PRIVATE src ${REF}/src)
      target_compile_definitions(app PRIVATE CODEC_LC3_LITE=1 LC3_LITE_PROFILE=1 BENCH_HAS_ASYNC=1)
    endif()
    target_sources(app PRIVATE ${CODEC_SOURCES})
    target_include_directories(app PRIVATE ${REF}/include)
    target_compile_definitions(app PRIVATE LC3_PLUS=1 LC3_PLUS_HR=1)
  endif()
  set_source_files_properties(${CODEC_SOURCES} src/codec.c src/codec_opus.c src/codec_sbc.c src/codec_wavpack.c PROPERTIES COMPILE_OPTIONS "-O3;-std=c11")
  if(BENCH_CODEC MATCHES "^google.*-lto$")
    set_property(SOURCE ${CODEC_SOURCES} APPEND PROPERTY COMPILE_OPTIONS -ffast-math)
  endif()
  if(BENCH_CODEC STREQUAL "google-custom-lto")
    # Experiment with static-loop expansion in the existing transform kernel.
    # Retain the transform, window and coefficient tables.
    set_property(SOURCE ${CUSTOM_DIR}/mdct.c APPEND PROPERTY COMPILE_OPTIONS -funroll-loops)
  endif()
  target_link_libraries(app PRIVATE m)
elseif(BENCH_ROLE STREQUAL "rx")
  target_sources(app PRIVATE src/rx_ipc.c)
  target_compile_definitions(app PRIVATE BENCH_RX=1)
endif()
